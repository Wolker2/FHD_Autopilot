                05=else if(owner==3)ownerFactor=1.15;
                06=else if(owner==4)ownerFactor=1.3;

            09=else if(difficulty==1)difficultyFactor=1.0;

            11=else difficultyFactor=1.5+(difficulty-3)*0.3;
            12=float timeFactor=1.0+9.0*(min(15000,max(1000,CurTurn()))-1000)/14000;
            13=result=round(amount*timeFactor*ownerFactor*difficultyFactor);
            14=if(result<0)result=0;
        }
